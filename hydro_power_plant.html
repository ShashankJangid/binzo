// HYDRO POWER 2.0 ESP32 PINOUT
#include <WiFi.h>
#include <FirebaseESP32.h>
#include <DHT.h>

#define DHTPIN 4
#define PUMP_RELAY 16
#define LIGHT_RELAY 17

DHT dht(DHTPIN, DHT11);
FirebaseData fbdo;

void setup() {
  Serial.begin(115200);
  pinMode(PUMP_RELAY, OUTPUT);
  pinMode(LIGHT_RELAY, OUTPUT);
  dht.begin();
}
